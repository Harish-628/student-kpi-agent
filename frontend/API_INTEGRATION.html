🔗 API Integration Guide

Complete guide for integrating the frontend with your FastAPI backend

📊 API Endpoints Overview

Endpoint Method Description
/student/add POST Add a new student
/student/{student_id} GET Get student information
/kpi/add POST Add KPI data for a student
/student/{student_id}/kpi GET Get student KPI data
/kpi/upload POST Bulk upload KPI data via CSV
/ GET Health check

📝 Step 3: API Request Methods

Add these methods to the Dashboard class:

// Fetch with auth headers async fetchAPI(endpoint, options = {}) { const token = this.getAuthToken(); const headers = { 'Content-Type': 'application/json', ...options.headers }; if (token) { headers['Authorization'] = `Bearer ${token}`; } const response = await fetch(`${this.API_BASE_URL}${endpoint}`, { ...options, headers }); if (!response.ok) { throw new Error(`API Error: ${response.status}`); } return response.json(); } // Add Student via API async addStudentViaAPI(studentData) { try { const result = await this.fetchAPI('/student/add', { method: 'POST', body: JSON.stringify(studentData) }); this.showToast('Student added successfully!', 'success'); return result; } catch (error) { this.showToast('Error adding student: ' + error.message, 'error'); } } // Get Student Data async getStudentViaAPI(studentId) { try { return await this.fetchAPI(`/student/${studentId}`); } catch (error) { this.showToast('Error fetching student: ' + error.message, 'error'); } } // Add KPI Data via API async addKPIViaAPI(kpiData) { try { const result = await this.fetchAPI('/kpi/add', { method: 'POST', body: JSON.stringify(kpiData) }); this.showToast('KPI data saved successfully!', 'success'); return result; } catch (error) { this.showToast('Error saving KPI: ' + error.message, 'error'); } } // Get Student KPI async getStudentKPIViaAPI(studentId) { try { return await this.fetchAPI(`/student/${studentId}/kpi`); } catch (error) { this.showToast('Error fetching KPI: ' + error.message, 'error'); } }

// Global error handler window.addEventListener('error', (event) => { console.error('Global error:', event.error); // Log to server for monitoring fetch('/api/logs', { method: 'POST', body: JSON.stringify({ message: event.error.message, stack: event.error.stack, timestamp: new Date().toISOString() }) }); }); // Handle fetch errors async function safeAPI(promise) { try { return await promise; } catch (error) { if (error instanceof TypeError) { console.error('Network error:', error); // Show offline message } else { console.error('API error:', error); } throw error; } }

📝 Authentication Token Management

Implement JWT token refresh mechanism:

// Token refresh helper async refreshToken() { try { const response = await fetch(this.API_BASE_URL + '/auth/refresh', { method: 'POST', headers: { 'Authorization': `Bearer ${this.getAuthToken()}` } }); if (response.ok) { const { token } = await response.json(); const user = JSON.parse(localStorage.getItem('user')); user.token = token; localStorage.setItem('user', JSON.stringify(user)); return token; } else { // Token expired, redirect to login window.location.href = 'index.html'; } } catch (error) { console.error('Token refresh failed:', error); } } // Use in fetchAPI async fetchAPI(endpoint, options = {}) { let token = this.getAuthToken(); const headers = { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}`, ...options.headers }; let response = await fetch(`${this.API_BASE_URL}${endpoint}`, { ...options, headers }); // If 401 (unauthorized), try to refresh token if (response.status === 401) { token = await this.refreshToken(); if (token) { headers['Authorization'] = `Bearer ${token}`; response = await fetch(`${this.API_BASE_URL}${endpoint}`, { ...options, headers }); } } if (!response.ok) { throw new Error(`API Error: ${response.status}`); } return response.json(); }
